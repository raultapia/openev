234 [[nodiscard]] inline double distance(const Event_<T> &e, const uint8_t type = DISTANCE_NORM_L2 | DISTANCE_FLAG_SPATIAL) const {
235 if(static_cast<bool>(type & DISTANCE_FLAG_SPATIOTEMPORAL)) {
236 return cv::norm(cv::Matx<T, 3, 1>(Event_<T>::x - e.x, Event_<T>::y - e.y, Event_<T>::t - e.t), type & 0x0FU);
237 }
238 if(static_cast<bool>(type & DISTANCE_FLAG_SPATIAL) || !static_cast<bool>(type & 0xF0U)) {
239 return cv::norm(cv::Matx<T, 2, 1>(Event_<T>::x - e.x, Event_<T>::y - e.y), type & 0x0FU);
240 }
241 if(static_cast<bool>(type & DISTANCE_FLAG_TEMPORAL)) {
242 return Event_<T>::t - e.t;
243 }
244 CV_Error(cv::Error::StsBadArg, "ev::Event_::distance: Bad distance option.");
