209 if(static_cast<bool>(type & DISTANCE_FLAG_SPATIOTEMPORAL)) {

211 }

213 return cv::norm(cv::Matx<T, 2, 1>(Event_<T>::x - e.x, Event_<T>::y - e.y), type & 0x0FU);
214 }
215 if(static_cast<bool>(type & DISTANCE_FLAG_TEMPORAL)) {
216 return Event_<T>::t - e.t;
