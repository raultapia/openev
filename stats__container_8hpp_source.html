30inline double cLog2C(const uint32_t c) {
31 static const std::array<double, TABULATED> table = [] {
32 std::array<double, TABULATED> t{};
33 for(std::size_t i = 1; i < TABULATED; i++) {
34 t[i] = static_cast<double>(i) * std::log2(static_cast<double>(i));

36 return t;
37 }();
