184 [[nodiscard]] inline cv::Matx<ResultType, 2, 2> covariance() const {
185 if constexpr(has_sum_<Container>::value && has_squares_<Container>::value) {
186 const auto n = static_cast<ResultType>(self_().size());
187 const cv::Point_<ResultType> mean = self_().sum() / n;
188 const cv::Vec<ResultType, 3> squares = self_().squares() / n;
189 const ResultType xy = squares[2] - mean.x * mean.y;
190 return {squares[0] - mean.x * mean.x, xy, xy, squares[1] - mean.y * mean.y};
191 } else {
192 const cv::Point_<ResultType> mean = meanPoint();
193 ResultType xx{0};
194 ResultType yy{0};
195 ResultType xy{0};
196 for(const Event_<T> &e : self_()) {
197 const ResultType dx = e.x - mean.x;
198 const ResultType dy = e.y - mean.y;
199 xx += dx * dx;
200 yy += dy * dy;
201 xy += dx * dy;
202 }
203 const auto n = static_cast<ResultType>(self_().size());
204 return {xx / n, xy / n, xy / n, yy / n};
205 }

307 int left = INT_MAX;
308 int right = INT_MIN;
309 int top = INT_MAX;
310 int bottom = INT_MIN;
311 for(const auto &item : range) {

314 right = std::max(right, pixel.x);
315 top = std::min(top, pixel.y);

321 template <typename Fn>
322 inline static void forEachPixelCount_(const std::vector<cv::Point> &pixels, Fn fn) {
323 constexpr uint64_t MAX_AREA_PER_EVENT = 32;
324 const cv::Rect bounds = bounds_(pixels, [](const cv::Point &pixel) { return pixel; });
325 const auto width = static_cast<uint64_t>(bounds.width);
326 const auto height = static_cast<uint64_t>(bounds.height);
327
328 if(width <= (MAX_AREA_PER_EVENT * pixels.size()) / height) {
329 std::vector<uint32_t> counts(width * height, 0);
330 for(const cv::Point &pixel : pixels) {
331 counts[(static_cast<uint64_t>(pixel.y - bounds.y) * width) + static_cast<uint64_t>(pixel.x - bounds.x)]++;
332 }
333 for(const uint32_t count : counts) {
334 if(count > 0) {
335 fn(count);
336 }
337 }
338 } else {
339 std::vector<uint64_t> keys;
340 keys.reserve(pixels.size());
341 for(const cv::Point &pixel : pixels) {
342 keys.push_back((static_cast<uint64_t>(static_cast<uint32_t>(pixel.y)) << 32U) | static_cast<uint32_t>(pixel.x));
343 }
344 std::sort(keys.begin(), keys.end());
345 uint32_t run = 1;
346 for(std::size_t i = 1; i <= keys.size(); i++) {
347 if(i == keys.size() || keys[i] != keys[i - 1]) {
348 fn(run);
349 run = 1;
350 } else {
351 run++;
352 }
353 }
354 }
355 }
